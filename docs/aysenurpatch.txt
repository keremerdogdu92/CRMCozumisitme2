Backend helper (Supabase API)

Yeni dosya: src/features/patients/api/api.cardFees.ts gibi:

export type CardFeeProfile = {
  id: string;
  bank_name: string;
  card_brand: string | null;
  installment_count: number;
  fee_rate_percent: number;
  valid_from: string;
  valid_to: string | null;
};

export async function fetchCurrentCardFeeForOrg(
  orgId: string,
  bankName: string,
  installmentCount: number
): Promise<CardFeeProfile | null> {
  const { data, error } = await supabaseClient
    .from('card_fee_profiles')
    .select('*')
    .eq('org_id', orgId)
    .eq('bank_name', bankName)
    .eq('installment_count', installmentCount)
    .lte('valid_from', new Date().toISOString())
    .or('valid_to.is.null,valid_to.gte.' + new Date().toISOString())
    .order('valid_from', { ascending: false })
    .limit(1)
    .maybeSingle();

  if (error) {
    console.error('CARD_FEE_FETCH:', error);
    return null;
  }
  return data as CardFeeProfile | null;
}
